{{- /* The physical object a note is drawn as: a frame of film, a TV, a paperback, or a record sleeve. Takes the note as "page" and "size", either "card" or "page". Its art is the note's cover when assets/covers/.* exists (`just cover` fetches film and TV art from TMDB), resized for where it shows. Otherwise it is a flat color field printed with one to three glyphs from static/glyphs.svg, in the manner of the Love, Death & Robots title cards. Unknown media, colors and glyphs fail the build either way, so a typo cannot ship a blank cover. */ -}} {{- $size := .size -}} {{- with .page -}} {{- $medium := .Params.medium -}} {{- $colors := slice "lime" "ink" "red" "cobalt" "tangerine" "paper" "pink" -}} {{- $color := .Params.color | default (index $colors (mod (hash.FNV32a .Title) (len $colors))) -}} {{- if not (in $colors $color) }}{{ errorf "%s: unknown color %q, want one of %v" .File.Path $color $colors }}{{ end -}} {{- $defaults := dict "film" (slice "reel") "tv" (slice "tv") "book" (slice "book") "music" (slice "note") -}} {{- $glyphs := .Params.glyphs | default (index $defaults $medium) -}} {{- if gt (len $glyphs) 3 }}{{ errorf "%s: at most three glyphs, got %d" .File.Path (len $glyphs) }}{{ end -}} {{- $sprite := readFile "static/glyphs.svg" -}} {{- $sheet := "glyphs.svg" | relURL -}} {{- $field := "" -}} {{- range $glyphs -}} {{- if not (strings.Contains $sprite (printf `id="%s"` .)) }}{{ errorf "%s: no glyph %q in static/glyphs.svg" $.File.Path . }}{{ end -}} {{- $field = printf `%s` $field $sheet . -}} {{- end -}} {{- $field = printf `%s` $color (len $glyphs) $field | safeHTML -}} {{- with resources.GetMatch (printf "covers/%s.*" .File.ContentBaseName) -}} {{- /* Rendered at 2x the widest the object gets in each place. */ -}} {{- $sm := .Resize "180x jpg q82" -}} {{- $lg := .Resize "360x jpg q82" -}} {{- $field = printf `` $sm.RelPermalink $sm.RelPermalink $lg.RelPermalink (cond (eq $size "page") "180px" "90px") | safeHTML -}} {{- end -}} {{- if eq $medium "film" -}} {{ $field }} {{- else if eq $medium "tv" -}} {{ $field }} {{- else if eq $medium "book" -}} {{ $field }} {{- else if eq $medium "music" -}} {{ $field }} {{- else -}} {{ errorf "%s: medium %q has no object; want film, tv, book, music, or note" .File.Path $medium }} {{- end -}} {{- end -}}
